class Tick { final double price; final int epoch, digit; Tick(this.price,this.epoch,this.digit); }
class Market {
  final String symbol, name; final List<Tick> ticks=[];
  bool live=false; DateTime? seen; int precision=2; String trend='COLLECTING'; double rsi=50, macd=0, atr=0, score=0; int? prediction; int? predictionEpoch; String predictionStrategy='Matches';
  Market(this.symbol,this.name);
  int? get lastDigit => ticks.isEmpty ? null : ticks.last.digit;
  List<int> get counts {final c=List<int>.filled(10,0); for(final t in ticks){c[t.digit]++;} return c;}
}
class Result {
  final String symbol, strategy; final int predicted, actual, epoch; final bool correct;
  Result(this.symbol,this.strategy,this.predicted,this.actual,this.epoch,this.correct);
  Map<String,dynamic> toJson()=>{'symbol':symbol,'strategy':strategy,'predicted':predicted,'actual':actual,'epoch':epoch,'correct':correct};
  factory Result.fromJson(Map<String,dynamic> j)=>Result(j['symbol'] as String,j['strategy'] as String,j['predicted'] as int,j['actual'] as int,j['epoch'] as int,j['correct'] as bool);
}
